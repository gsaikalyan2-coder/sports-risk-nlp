# Heavy ML dependency layer (Phase 13 onward: transformer fine-tuning,
# explainability, experiment tracking).
#
# INSTALL NOTE — torch must come from the PyTorch CPU wheel index, not PyPI.
# The default PyPI wheel for Linux is the CUDA build and drags in several GB of
# nvidia-* runtime libraries this project never uses (no GPU in the container).
# The owner's Windows venv already runs torch 2.13.0+cpu, which is correct.
#
# Host (Windows / Linux, CPU-only):
#     pip install -r requirements-ml.txt --extra-index-url https://download.pytorch.org/whl/cpu
#
# Container: Dockerfile.train already passes the CPU index.
# If you later train on a CUDA GPU, drop the --extra-index-url and let pip
# resolve the default (CUDA) wheel instead.

-r requirements-base.txt

torch>=2.2

# transformers 5.x is a breaking major over 4.x (tokenizer and Trainer API
# changes). src/models/ from Phase 13 targets 5.x; the bound stops a silent
# downgrade to 4.x resolving underneath it.
transformers>=5.0,<6.0
datasets>=5.0
accelerate>=1.0

# REQUIRED for microsoft/deberta-v3-base, which is the Phase 14 default base
# model (config/settings.yaml -> model.base). DeBERTa-v3 ships its vocabulary as
# a SentencePiece model (spm.model), and transformers 5.x needs this package to
# convert it into a fast tokenizer.
#
# It is pinned here because omitting it does NOT produce a clean "missing
# dependency" error. transformers catches the ImportError, falls back to a
# TikToken extractor, and that extractor tries to parse the SentencePiece
# protobuf as a text BPE file -- so the surfaced error is an unrelated-looking
# `ValueError: Error parsing line b'\x0e'` from deep inside tiktoken, several
# frames away from the actual cause. Encountered on first run, 2026-08-11
# (OPEN-029). roberta-base does not need it (byte-level BPE), which is why
# --base-model roberta-base is the documented escape hatch.
sentencepiece>=0.2

# Explainability & tracking
shap>=0.46
wandb>=0.17
